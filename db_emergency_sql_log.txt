UPDATE atm_update a
LEFT JOIN atm_master m ON a.atm_id = m.atm_id
LEFT JOIN vendor_master av ON m.atm_vendor_id = av.id
LEFT JOIN vendor_master uv ON m.ups_vendor_id = uv.id
LEFT JOIN problem_master p ON a.problem = p.problem_name
SET a.responsible_vendor_name = 
    CASE 
        WHEN UPPER(p.responsible_vendor_type) = 'UPS' THEN COALESCE(uv.vendor_name, m.ups_vendor, '')
        WHEN UPPER(p.responsible_vendor_type) IN ('ATM', 'CRM') THEN COALESCE(av.vendor_name, m.atm_vendor, '')
        WHEN p.responsible_vendor_type IS NOT NULL AND TRIM(p.responsible_vendor_type) != '' THEN p.responsible_vendor_type
        WHEN COALESCE(av.vendor_name, m.atm_vendor, '') != '' THEN COALESCE(av.vendor_name, m.atm_vendor, '')
        ELSE COALESCE(uv.vendor_name, m.ups_vendor, '')
    END
WHERE a.incident_status = 'Open' 
AND (a.responsible_vendor_name IS NULL OR TRIM(a.responsible_vendor_name) = '');
